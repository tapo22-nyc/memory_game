-- ============================================================
-- TABLE 1: closest_call_matches
-- ============================================================
CREATE TABLE IF NOT EXISTS closest_call_matches (
  id                 SERIAL PRIMARY KEY,
  fantasy_match_id   INT    REFERENCES fantasy_matches(id) ON DELETE SET NULL,
  ipl_match_id       INT    REFERENCES ipl_matches(id)     ON DELETE SET NULL,
  match_title        TEXT   NOT NULL,
  team_1             TEXT   NOT NULL,
  team_2             TEXT   NOT NULL,
  match_date         TIMESTAMP WITH TIME ZONE,
  status             TEXT   NOT NULL DEFAULT 'open'
                       CHECK (status IN ('open','locked','closed')),
  created_at         TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at         TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);


-- ============================================================
-- TABLE 2: closest_call_questions
-- ============================================================
CREATE TABLE IF NOT EXISTS closest_call_questions (
  id                     SERIAL PRIMARY KEY,
  closest_call_match_id  INT  NOT NULL REFERENCES closest_call_matches(id) ON DELETE CASCADE,
  question_text          TEXT NOT NULL,
  question_type          TEXT NOT NULL,
  -- Examples: team1_score_6, team1_score_12, team1_final_score,
  --           team2_score_6, team2_score_12, team2_final_score,
  --           batter_runs, bowler_wickets, total_sixes, total_wickets
  phase                  TEXT NOT NULL,
  -- Examples: pre_match, powerplay, mid_innings, final
  team_code              TEXT,
  player_id              INT,
  player_name            TEXT,
  innings_number         INT,
  max_points             INT  NOT NULL DEFAULT 100,
  scoring_rule_code      TEXT NOT NULL,
  -- Examples: TEAM_PHASE_SCORE, TEAM_FINAL_SCORE, BATTER_RUNS, BOWLER_WICKETS
  open_time              TIMESTAMP WITH TIME ZONE,
  lock_time              TIMESTAMP WITH TIME ZONE,
  status                 TEXT NOT NULL DEFAULT 'open'
                           CHECK (status IN ('open','locked','scored','void')),
  actual_value           INT,
  created_at             TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at             TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);


-- ============================================================
-- TABLE 3: closest_call_predictions
-- ============================================================
CREATE TABLE IF NOT EXISTS closest_call_predictions (
  id                     SERIAL PRIMARY KEY,
  user_id                INT    NOT NULL REFERENCES ipl_users(id) ON DELETE CASCADE,
  closest_call_match_id  INT    NOT NULL REFERENCES closest_call_matches(id) ON DELETE CASCADE,
  question_id            INT    NOT NULL REFERENCES closest_call_questions(id) ON DELETE CASCADE,
  predicted_value        INT    NOT NULL,
  actual_value           INT,
  difference             INT,
  points_awarded         NUMERIC(8,2) DEFAULT 0,
  submitted_at           TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at             TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE (user_id, question_id)
);


-- ============================================================
-- TABLE 4: closest_call_score_snapshots
-- Purpose: manually store 6-over, 12-over, and final scores
-- ============================================================
CREATE TABLE IF NOT EXISTS closest_call_score_snapshots (
  id                     SERIAL PRIMARY KEY,
  closest_call_match_id  INT  NOT NULL REFERENCES closest_call_matches(id) ON DELETE CASCADE,
  fantasy_match_id       INT  REFERENCES fantasy_matches(id) ON DELETE SET NULL,
  innings_number         INT  NOT NULL,
  batting_team           TEXT NOT NULL,
  phase                  TEXT NOT NULL,
  -- Examples: six_overs, twelve_overs, final
  overs                  NUMERIC(4,1),
  runs                   INT  NOT NULL,
  wickets                INT,
  source                 TEXT NOT NULL DEFAULT 'manual',
  -- Examples: manual, cricapi
  created_at             TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at             TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE (closest_call_match_id, innings_number, phase)
);


-- ============================================================
-- TABLE 5: closest_call_scoring_rules
-- Purpose: configurable point brackets per rule code
-- ============================================================
CREATE TABLE IF NOT EXISTS closest_call_scoring_rules (
  id         SERIAL PRIMARY KEY,
  rule_code  TEXT NOT NULL,
  min_diff   INT  NOT NULL,
  max_diff   INT,           -- NULL means "infinity" (catch-all for this bracket)
  points     INT  NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
